### Task 5.2: Python as a calculator (checked in Python, expression -> result)
>>> bin(13)
'0b1101'
>>> bin(25)
'0b11001'
>>> int("1010", 2)
10
>>> int("1101", 2)
13
>>> int("A", 16)
10
>>> int("1F", 16)
31
>>> hex(int("1111", 2))
'0xf'
>>> hex(10)
'0xa'
>>> hex(31)
'0x1f'

### Task 5.4: text encoding
ord('A') = 65
ord('a') = 97
chr(65) = A
'A'.encode('ascii') = b'A'
ord('₹') = 8377 (Unicode code point U+20B9)
'₹'.encode('utf-8') = b'\xe2\x82\xb9' -> 3 bytes
'₹'.encode('ascii') fails: UnicodeEncodeError
